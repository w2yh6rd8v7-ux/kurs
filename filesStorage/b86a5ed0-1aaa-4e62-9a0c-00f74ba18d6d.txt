Подключаем необходимые атрибуты контроллера и описываем метод вывода списка сотрудников по примеру выбора списка отделов. Указываем, что помимо ФИО, почты, даты рождения и номера телефона нам необходимо получать информацию об опыте работы и образовании.

[Рисунок 7 -- Файл EmployeesController.cs (фрагмент метода GetEmployees)]
[ApiController]
[Route("api/[controller]")]
public class EmployeesController : ControllerBase
{
    private readonly ApplicationDbContext _context;
    public EmployeesController(ApplicationDbContext context)
    {
        _context = context;
    }

    [HttpGet]
    public async Task<IActionResult> GetEmployees()
    {
        var employees = await _context.Employees
            .Include(e => e.WorkExperiences)
            .Include(e => e.Educations)
            .Select(e => new
            {
                e.Id,
                e.FullName,
                e.Email,
                e.BirthDate,
                e.PhoneNumber,
                WorkExperiences = e.WorkExperiences.ToList(),
                Educations = e.Educations.ToList()
            })
            .ToListAsync();
        return Ok(employees);
    }
}

[HttpPut("{id}")]
public async Task<IActionResult> UpdateEmployee(int id, [FromBody] EmployeeUpdateDto dto)

В методе DeleteEmployee мы будем искать сотрудника с введенным id и в случае успешного обнаружения, удалять его. В методе AddWorkExperience, находим сотрудника по id и получаем его данные с помощью ранее созданного класса dto. Переменная weId отвечает за получение идентификатора вновь созданной записи.

[Рисунок 9 -- Методы DeleteEmployee и AddWorkExperience]
[HttpDelete("{id}")]
public async Task<IActionResult> DeleteEmployee(int id)
{
    var employee = await _context.Employees.FindAsync(id);
    if (employee == null)
        return NotFound();
    _context.Employees.Remove(employee);
    await _context.SaveChangesAsync();
    return Ok();
}

[HttpPost("AddWorkExperience")]
public async Task<IActionResult> AddWorkExperience([FromBody] WorkExperienceRequestDto dto)
{
    var employee = await _context.Employees.FindAsync(dto.EmployeeId);
    if (employee == null)
        return NotFound("Сотрудник не найден");
    var workExp = new WorkExperience
    {
        EmployeeId = dto.EmployeeId,
        Company = dto.Company,
        Position = dto.Position,
        StartDate = dto.StartDate,
        EndDate = dto.EndDate
    };
    _context.WorkExperiences.Add(workExp);
    await _context.SaveChangesAsync();
    int weId = workExp.Id; // переменная weId отвечает за получение идентификатора
    return Ok(workExp);
}

[Рисунок 20 -- Метод Register]
[HttpPost("register")]
public async Task<IActionResult> Register([FromBody] RegisterRequestDto request)
{
    var existingUser = await _context.Users.FirstOrDefaultAsync(u => u.Email == request.Email);
    if (existingUser != null)
        return BadRequest("Пользователь с таким email уже существует");

    var user = new User
    {
        Email = request.Email,
        PasswordHash = HashPassword(request.Password),
        FullName = request.FullName
    };
    _context.Users.Add(user);
    await _context.SaveChangesAsync();
    return Ok();
}
